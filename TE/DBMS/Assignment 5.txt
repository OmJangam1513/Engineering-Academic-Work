CREATE OR REPLACE FUNCTION get_student_category(p_marks IN NUMBER)
RETURN VARCHAR2
IS
    v_category VARCHAR2(30);
BEGIN
    IF p_marks BETWEEN 1000 AND 1499 THEN
        v_category := 'Distinction';
    ELSIF p_marks BETWEEN 900 AND 989 THEN
        v_category := 'First Class';
    ELSIF p_marks BETWEEN 825 AND 899 THEN
        v_category := 'Higher Second Class';
    ELSE
        v_category := 'No Category';
    END IF;

    RETURN v_category;
END;
/


CREATE OR REPLACE PROCEDURE categorize_all_students
IS
    
    CURSOR c_student IS
        SELECT roll_no, marks FROM student;

    
    v_category VARCHAR2(30);
BEGIN
    
    FOR rec IN c_student LOOP
       
        v_category := get_student_category(rec.marks);

        
        UPDATE student
        SET category = v_category
        WHERE roll_no = rec.roll_no;

        DBMS_OUTPUT.PUT_LINE('Roll No: ' || rec.roll_no ||
                             ' | Marks: ' || rec.marks ||
                             ' | Category: ' || v_category);
    END LOOP;

    COMMIT;

    DBMS_OUTPUT.PUT_LINE('All students categorized successfully.');
EXCEPTION
    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE('Error: ' || SQLERRM);
END;
/
